# Optional dependencies for the Sarvam-1 correction engines
# (--engine sarvam-rerank / hybrid / sarvam-generate).
#
# Kept out of requirements.txt on purpose: this pulls ~7 GB of CUDA wheels,
# and the rule engine -- still the default -- needs none of it.
#
# Install the torch build that matches your driver. `pip install torch` grabs
# the newest CUDA build, which will fail at runtime with "NVIDIA driver on your
# system is too old" on anything older. Check with `nvidia-smi` and use the
# matching index, e.g. for a CUDA 12.8 driver (535.x-57x):
#
#   pip install torch --index-url https://download.pytorch.org/whl/cu128
#   pip install -r requirements-sarvam.txt
#
# Model weights (~4.3 GB) download from Hugging Face on first use and are
# cached in ~/.cache/huggingface. sarvamai/sarvam-1 is gated: accept the
# licence on the model page and `huggingface-cli login` first.

torch>=2.4.0
transformers>=4.44.0
accelerate>=0.33.0
sentencepiece>=0.2.0

# Only needed for sarvam-30b-fp8. Sarvam's own fp8 build is quantized with
# NVIDIA ModelOpt, a format transformers has no quantizer for -- use RedHat
# AI's compressed-tensors rebuild of the same fp8 weights instead
# (KNOWN_MODELS['sarvam-30b-fp8'] already points there).
